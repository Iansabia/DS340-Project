def alpha_bps_per_trade(total_pnl: float, num_trades: int, position_size: float = CANONICAL_POSITION_SIZE) -> float:
    """Per-trade alpha expressed in basis points.

    Formula: (total_pnl / num_trades / position_size) * 10_000
    Example: LR with $201.69 P&L over 1549 trades at $100 position size:
      201.69 / 1549 / 100 * 10000 = 13.02 bps per trade.
    """
    if num_trades <= 0:
        return 0.0
    return (total_pnl / num_trades / position_size) * 10_000.0


def evaluate_model(name: str, predictions: np.ndarray, actuals: np.ndarray,
                   timestamps: np.ndarray, test_df: pd.DataFrame) -> dict:
    """Compute every canonical metric for a single model from its predictions."""
    profit = simulate_profit(predictions, actuals,
                             threshold=CANONICAL_THRESHOLD,
                             timestamps=timestamps)
    bt = WalkForwardBacktester(position_size=CANONICAL_POSITION_SIZE,
                               threshold=CANONICAL_THRESHOLD).run(test_df, predictions)
    rmse = float(np.sqrt(((predictions - actuals) ** 2).mean()))
    mae = float(np.abs(predictions - actuals).mean())
    dir_acc = float((np.sign(predictions) == np.sign(actuals)).mean())
    return {
        "model": name,
        "rmse": rmse,
        "mae": mae,
        "directional_accuracy": dir_acc,
        "total_pnl": profit["total_pnl"],
        "num_trades": profit["num_trades"],
        "win_rate": profit["win_rate"],
        "sharpe_per_trade": profit["sharpe_per_trade"],
        "sharpe_annualized": profit["sharpe_ratio"],   # daily-aggregated, sqrt(365)
        "alpha_bps_per_trade": alpha_bps_per_trade(profit["total_pnl"], profit["num_trades"]),
        "max_drawdown_pct": bt.get("max_drawdown", 0.0),
        "position_size_usd": CANONICAL_POSITION_SIZE,
        "threshold": CANONICAL_THRESHOLD,
        "seed": CANONICAL_SEED,
