      local name="$1"; local expected="$2"; local actual="$3"

        printf "  [OK]   %-50s (got %s)\n" "$name" "$actual"
      else
        printf "  [FAIL] %-50s (want %s, got %s)\n" "$name" "$expected" "$actual"

        printf "  [FAIL] %-50s (got %s, want >= %s)\n" "$name" "$actual" "$minimum"
