    Step A — Create experiments/run_tft.py (~80 LOC):

    and implements the Option B gate: always produces an output artifact.

    Structure:
    ```python
    """Single-split TFT experiment runner.

    Runs TFTPredictor with 3 seeds on the standard single-split (train.parquet / test.parquet).
    Implements Option B gate: always writes TFT.json — either with real results or a
    documented negative-result sentinel when val_loss >= GRU baseline (0.2928 RMSE).

    Usage:
        python -m experiments.run_tft
    """
    import json, sys, time, traceback, warnings
    from pathlib import Path
    import numpy as np
    import pandas as pd

    # Add project root to sys.path (same pattern as run_baselines.py)
    sys.path.insert(0, str(Path(__file__).parent.parent))

    from src.utils.seed import set_all_seeds
    from src.features.compute_features import compute_derived_features
    from src.models.tft import TFTPredictor
    from experiments.run_baselines import prepare_xy_for_seq, THRESHOLD

    GRU_RMSE_BASELINE = 0.2928
    SEEDS = [42, 7, 123]
    RESULTS_DIR = Path("experiments/results/tier2")

    def main():
        set_all_seeds(42)
        RESULTS_DIR.mkdir(parents=True, exist_ok=True)

        train = pd.read_parquet("data/processed/train.parquet")
        test = pd.read_parquet("data/processed/test.parquet")
        train = compute_derived_features(train)
        test = compute_derived_features(test)
        feature_cols = [c for c in train.select_dtypes(include='number').columns
                        if c not in ('target', 'time_idx')]

        X_train, y_train = prepare_xy_for_seq(train, feature_cols)
        X_test, y_test = prepare_xy_for_seq(test, feature_cols)

        seed_rmses, seed_pnls = [], []
        last_metrics, last_audit = None, None

        for seed in SEEDS:
            print(f"\n{'='*60}\nTraining TFT seed={seed}\n{'='*60}")
            t0 = time.time()
            try:
                model = TFTPredictor(random_state=seed)
                model.fit(X_train, y_train)
                metrics = model.evaluate(X_test, y_test, threshold=THRESHOLD)
                metrics.pop("pnl_series", None)
                audit = model._audit_attention()
                print(f"  RMSE={metrics['rmse']:.4f}  P&L=${metrics['total_pnl']:.2f}  "
                      f"entropy={audit['entropy']:.3f}  degenerate={audit['is_degenerate']}")
                seed_rmses.append(metrics["rmse"])
                seed_pnls.append(metrics["total_pnl"])
                last_metrics = metrics
                last_audit = audit
            except Exception as e:
                print(f"  FAILED: {e}")
                traceback.print_exc()
                # Write negative result artifact
                negative = {
                    "model": "TFT",
                    "converged": False,
                    "failure_reason": str(e),
                    "note": "TFT did not converge at N=6802. Documented as negative result per TFT-04 Option B.",
                    "gru_baseline_rmse": GRU_RMSE_BASELINE,
                }
                out = RESULTS_DIR / "TFT.json"
                out.write_text(json.dumps(negative, indent=2))
                print(f"\nNegative result written to {out}")
                return

        avg_rmse = float(np.mean(seed_rmses))
        converged = avg_rmse < GRU_RMSE_BASELINE
        result = {
            "model": "TFT",
            "converged": converged,
            "rmse": avg_rmse,
            "rmse_std": float(np.std(seed_rmses)),
            "seed_rmses": seed_rmses,
            "total_pnl": float(np.mean(seed_pnls)),
            "gru_baseline_rmse": GRU_RMSE_BASELINE,
            "beats_gru": converged,
            "attention_audit": last_audit,
            **{k: v for k, v in (last_metrics or {}).items()},
            "note": (
                "TFT converged and beats GRU baseline." if converged
                else "TFT did not beat GRU at N=6802. Documented negative result per TFT-04 Option B."
            ),
        }
        out = RESULTS_DIR / "TFT.json"
        out.write_text(json.dumps(result, indent=2))
        print(f"\nResult written to {out}")
        print(f"Converged: {converged} | Avg RMSE: {avg_rmse:.4f} | GRU baseline: {GRU_RMSE_BASELINE}")

    if __name__ == "__main__":
        main()
    ```
