    ```python
    class EnsemblePredictor(BasePredictor):
        def __init__(
            self,
            members: list[BasePredictor],
            weights: list[float] | None = None,
            concordance_mode: str = "none",  # "none" | "strict"
            seed: int = 42,
        ) -> None:
    ```
    - Raise ValueError if members is empty.
    - Raise ValueError if weights sums to 0.
    - Store as self._members, self._weights (np.array float64), self._concordance_mode, self._seed.
    - self._fitted = False.

    name property:
    ```python
    @property
    def name(self) -> str:
        member_names = "+".join(m.name for m in self._members)
        return f"Ensemble({member_names}, {self._concordance_mode})"
    ```

    fit():
    ```python
    def fit(self, X_train: pd.DataFrame, y_train: np.ndarray) -> "EnsemblePredictor":
        set_all_seeds(self._seed)
        for m in self._members:
            m.fit(X_train, y_train)
        self._fitted = True
        return self
    ```

    predict():
    ```python
    def predict(self, X: pd.DataFrame) -> np.ndarray:
        if not self._fitted:
            raise RuntimeError("EnsemblePredictor must be fit before predict")
        preds = np.stack([m.predict(X) for m in self._members])  # (n_members, n_rows)
        weights = self._weights / self._weights.sum()             # normalize
        weighted = (weights[:, None] * preds).sum(axis=0)        # (n_rows,)
        if self._concordance_mode == "strict":
            signs = np.sign(preds)
            agree = np.all(signs == signs[[0], :], axis=0)       # (n_rows,) bool
            return np.where(agree, weighted, 0.0)
        return weighted
    ```

    Picklability: No custom __getstate__/__setstate__ needed. pickle.dump(self, f) in BasePredictor.save() handles the full object including _members list. All member predictors (LR, XGB, LSTM) are picklable.
